================================================================================
Real-world rate-limiter pattern with table operations
================================================================================

# Adapted from clouddocs.f5.com/api/irules/HTTP-URI-Request-Limiter.html.
# Original used word operators (not, and); rewritten with symbolic forms
# (!, &&) to avoid the documented grammar gap. See README "Known limitations".
when HTTP_REQUEST {
    set ip_addr [IP::client_addr]
    set tbl "limit_$ip_addr:[HTTP::host][HTTP::uri]"

    set banTime [table lookup -notouch -subtable $tbl banTime]
    if { $banTime > 0 } {
        log local0. "BANNED: $tbl, banTime=$banTime"
        drop
        event disable all
        return
    }

    set requestCnt [table lookup -notouch -subtable $tbl requestCnt]
    if { $requestCnt == "" } {
        table set -subtable $tbl requestCnt 1 10 10
    } else {
        set requestCnt [table incr -notouch -subtable $tbl requestCnt]
        if { $requestCnt > 20 } {
            table delete -subtable $tbl requestCnt
            table set -subtable $tbl banTime 10 10
        }
    }
}
